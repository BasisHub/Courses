lib/behaviour.js
  export function behaviourTest(id, title, opts, body)
    id: manifest id (e.g. 'dwc/01_GUI2BUI2DWC/DWC2.bbj') or fixture id ('fixture/known-good.bbj')
    opts: { issue?: 3|4|5|6|7, startDialog?: { button } | null, expectFailure?: boolean, deadlineMs?: number }   // deadlineMs only for the wrapper self-test
    body: async ({ page, app, note }) => void   // use Playwright expect; throw = fail; note(text) adds evidence
    Test title: `[${id}] issue #${issue}: ${title}` when issue is set, else `[${id}] ${title}`.
  export class BlockedError extends Error  // throw new BlockedError(reason) from a body when the check cannot be made (for example the oracle element is missing); outcome blocked, evidence = reason
    Outcome: blocked when the start state is not the kind's expected state (window/starter -> 'window', print -> 'console', dialog -> 'dialog'); category = the first classify reason category ('license' when no license was free), evidence = 'sample did not start (<state>): <classify reasons>'. Else pass (category null) when the body returns, blocked with category 'check' when it throws a BlockedError (evidence = its message), fail with category 'assertion' when it throws anything else (evidence 'failed: <first line of the error>'); a BBj !ERROR in the console after the body turns the outcome into fail with category 'bbj' and '!ERROR=<code> <message> at line <line>'. Deadline (W6): start, body and post-checks run inside withDeadline(opts.deadlineMs ?? deadlineFor(testInfo), ...); when the deadline passes first the outcome is fail with category 'timeout', evidence 'timed out after <s> s; last note: <text>; console: <excerpt joined by " / ">' and screenshot = timeoutEvidence(...).screenshot. Evidence is redacted. A screenshot `<app>--<slug(title) cut to 40>` is taken after the body. The result is written in a finally block (an error inside the wrapper itself leaves a fail result with category 'harness' and harnessIssue 'wrapper-error'), to results/behaviour (results/selftest-behaviour for fixture ids), and registered with bbj.recordResult(path). The test then asserts outcome === 'pass' (red tests are findings; a timeout is a finding, not a harness failure).
  export function smokeTest(app)                          // defines test(`smoke [${app.id}]`, ...): withDeadline(deadlineFor(testInfo), () => runProgram(bbj, app)); deadline -> makeSampleResult with status 'fail', reasons [{ category: 'timeout', detail }] and screenshots.light = the timeout screenshot; a thrown error -> status 'fail', reasons [{ category: 'start', detail: 'error: <first line>' }]; written in a finally block with writeSampleResult and bbj.recordResult; then expect(r.status, reasons as message).toBe('pass')
  export async function withDeadline(ms, fn)               // races fn() against a timer -> { timedOut: true } | { timedOut: false, value } | { timedOut: false, error }; clears the timer; a late rejection of fn after the deadline is caught and ignored
  export function deadlineFor(testInfo, marginMs = 15000)  // Math.max(5000, testInfo.timeout - marginMs), measured from the start of the test body
  export async function timeoutEvidence(page, base)       // each step capped at 5 s: page.screenshot to results/screenshots/<base>-timeout.png, readConsole(page, { waitForReadyMs: 0 }) -> redacted consoleExcerpt(parseConsole(lines)) -> { screenshot: path relative to resultsDir() | null, excerpt: string[] }
