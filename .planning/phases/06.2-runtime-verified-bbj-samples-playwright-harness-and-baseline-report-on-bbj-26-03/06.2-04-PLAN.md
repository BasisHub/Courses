Contract this plan creates:
lib/compile.js
  Rec = { file: string, severity: 'error'|'warning', typecheck: boolean, message: string, compiledLine: number, line: number, statement: string }
  export function parseCompileOutput(text)   // -> { records: Rec[], other: string[] }. Split on /\r?\n/. A line starts a record when it matches /^\S+\.bbj: /; every following line that does not start a record is appended (with '\n') to the open record until the record text contains /\] at line \d+ \(\d+\): / or, for syntax errors, / at line \d+ \(\d+\): /. Each record text is matched with /^(.+?): (type check )?(error|warning)(?: \[([\s\S]*?)\])? at line (\d+) \((\d+)\): (.*)$/. Non-empty lines outside records go to `other`.
  export function isFollowUp(rec)              // message contains '<undeclared>' or starts with 'Undeclared variable'
  export function summarizeTypecheck(records)  // -> { errors, warnings, undeclared, findings }; undeclared counts follow-ups; findings = error records that are not follow-ups, mapped to { line, severity, message, statement }
