  export const NOTE_SECTIONS = ['Static findings', 'Draft issues', 'Questions for Hyyan', 'Notes']
  export function escapeCell(text)      // '|' -> '\|', newlines -> ' ', trims, max 160 chars with '...' suffix
  export function splitNotes(md)        // -> { 'Static findings': string, 'Draft issues': string, 'Questions for Hyyan': string, 'Notes': string } (missing -> ''); throws on any other H2 heading
